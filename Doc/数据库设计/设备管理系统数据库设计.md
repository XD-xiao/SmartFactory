### 1. device_device（设备表）

| 字段名                | 类型           | 说明                   |
| ------------------ | ------------ | -------------------- |
| id                 | bigint       | 主键，自增                |
| sn                 | varchar(255) | 设备唯一标识               |
| type               | varchar(255) | 设备类型。pad、pda、adapter |
| state              | int          | 状态。0离线；1在线           |
| last\_online\_time | datetime     | 最后一次上报上线的时间          |
| create\_time       | datetime     | 创建时间                 |
| update\_time       | datetime     | 更新时间                 |

### 2. device_group（分组表）

| 字段名          | 类型           | 说明    |
| ------------ | ------------ | ----- |
| id           | bigint       | 主键，自增 |
| name         | varchar(255) | 组名    |
| create\_time | datetime     | 创建时间  |
| update\_time | datetime     | 更新时间  |

注意：当前分组没有任何设备加入，才可以删除分组

### 3. device_group_device（分组-设备关联表）

| 字段名          | 类型           | 说明     |
| ------------ | ------------ | ------ |
| id           | bigint       | 主键，自增  |
| group\_id    | bigint       | 分组id   |
| device\_sn   | varchar(255) | 设备唯一编码 |
| create\_time | datetime     | 创建时间   |

注意：一个设备只可以加入一个分组，一个分组可以有多个设备

### 4. device_machine_failure（设备故障表）

| 字段名                    | 类型           | 说明     |
| ---------------------- | ------------ | ------ |
| id                     | bigint       | 主键，自增  |
| sn                     | varchar(255) | 设备唯一标识 |
| reason                 | varchar(255) | 故障原因   |
| failure\_level         | int          | 故障级别   |
| failure\_time          | varchar(255) | 故障时间   |
| status                 | int          | 状态     |
| failure\_recover\_time | varchar(255) | 故障恢复时间 |
| create\_time           | datetime     | 创建时间   |

### 5. device_equipment_run_time（产线设备运行数据表）

| 字段名               | 类型          | 说明        |
| ----------------- | ----------- | --------- |
| id                | bigint      | 主键，自增     |
| sn                | varchar(255) | 设备sn码     |
| seq_no            | int       | 序号        |
| action            | int       | 当前状态 0关1开 |
| work\_begin\_time | datetime  | 开始工作时间    |
| work\_end\_time   | datetime  | 结束工作时间    |
| work\_time        | float     | 工作时长      |

```mysql
-- 建表语句
CREATE TABLE `device_equipment_run_time` (
    `id` BIGINT NOT NULL AUTO_INCREMENT COMMENT '主键，自增',
    `sn` VARCHAR(255) NOT NULL COMMENT '设备sn码',
    `seq_no` INT DEFAULT NULL COMMENT '序号',
    `action` INT NOT NULL DEFAULT 0 COMMENT '当前状态 0关1开',
    `work_begin_time` DATETIME DEFAULT NULL COMMENT '开始工作时间',
    `work_end_time` DATETIME DEFAULT NULL COMMENT '结束工作时间',
    `work_time` FLOAT DEFAULT NULL COMMENT '工作时长',
    PRIMARY KEY (`id`),
    KEY `idx_sn` (`sn`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='产线设备运行数据表';
```

## 数据库创建 SQL

```sql
-- 1. 设备表
CREATE TABLE `device_device` (
    `id` BIGINT NOT NULL AUTO_INCREMENT COMMENT '主键，自增',
    `sn` VARCHAR(255) NOT NULL COMMENT '设备唯一标识',
    `type` VARCHAR(255) NOT NULL COMMENT '设备类型。pad、pda、adapter',
    `state` INT NOT NULL DEFAULT 0 COMMENT '状态。0离线；1在线',
    `last_online_time` DATETIME DEFAULT NULL COMMENT '最后一次上报上线的时间',
    `create_time` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    `update_time` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
    PRIMARY KEY (`id`),
    UNIQUE KEY `uk_sn` (`sn`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='设备表';

-- 2. 分组表
CREATE TABLE `device_group` (
    `id` BIGINT NOT NULL AUTO_INCREMENT COMMENT '主键，自增',
    `name` VARCHAR(255) NOT NULL COMMENT '组名',
    `create_time` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    `update_time` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
    PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='分组表';
-- 注意：当前分组没有任何设备加入，才可以删除分组

-- 3. 分组-设备关联表
CREATE TABLE `device_group_device` (
    `id` BIGINT NOT NULL AUTO_INCREMENT COMMENT '主键，自增',
    `group_id` BIGINT NOT NULL COMMENT '分组id',
    `device_sn` VARCHAR(255) NOT NULL COMMENT '设备唯一编码',
    `create_time` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    PRIMARY KEY (`id`),
    UNIQUE KEY `uk_device_sn` (`device_sn`),
    KEY `idx_group_id` (`group_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='分组-设备关联表';
-- 注意：一个设备只可以加入一个分组，一个分组可以有多个设备

-- 4. 设备故障表
CREATE TABLE `device_machine_failure` (
    `id` BIGINT NOT NULL AUTO_INCREMENT COMMENT '主键，自增',
    `sn` VARCHAR(255) NOT NULL COMMENT '设备唯一标识',
    `reason` VARCHAR(255) DEFAULT NULL COMMENT '故障原因',
    `failure_level` INT DEFAULT NULL COMMENT '故障级别',
    `failure_time` VARCHAR(255) DEFAULT NULL COMMENT '故障时间',
    `status` INT DEFAULT NULL COMMENT '状态',
    `failure_recover_time` VARCHAR(255) DEFAULT NULL COMMENT '故障恢复时间',
    `create_time` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    PRIMARY KEY (`id`),
    KEY `idx_sn` (`sn`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='设备故障表';

-- 5. 产线设备运行数据表
CREATE TABLE `device_equipment_run_time` (
    `id` BIGINT NOT NULL AUTO_INCREMENT COMMENT '主键，自增',
    `sn` VARCHAR(255) NOT NULL COMMENT '设备sn码',
    `seq_no` INT DEFAULT NULL COMMENT '序号',
    `action` INT NOT NULL DEFAULT 0 COMMENT '当前状态 0关1开',
    `work_begin_time` DATETIME DEFAULT NULL COMMENT '开始工作时间',
    `work_end_time` DATETIME DEFAULT NULL COMMENT '结束工作时间',
    `work_time` FLOAT DEFAULT NULL COMMENT '工作时长',
    PRIMARY KEY (`id`),
    KEY `idx_sn` (`sn`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='产线设备运行数据表';
```
