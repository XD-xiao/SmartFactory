## 建表 SQL

```sql
-- 传感器配置表
CREATE TABLE `edubox_sensor_config` (
  `id`          BIGINT   NOT NULL AUTO_INCREMENT COMMENT '主键，自增',
  `sn`          VARCHAR(64) DEFAULT NULL            COMMENT '设备sn',
  `sensor`      BIGINT      DEFAULT NULL            COMMENT '传感器类型',
  `sensor_name` VARCHAR(64) DEFAULT NULL            COMMENT '传感器名称',
  `sceneType`   BIGINT      DEFAULT NULL            COMMENT '场景',
  `valueType`   TINYINT(1) DEFAULT NULL          COMMENT '取值类型: 0-范围; 1-开关',
  `max`         INT      DEFAULT NULL            COMMENT '传感器上限值',
  `min`         INT      DEFAULT NULL            COMMENT '传感器下限值',
  `max_th`      INT      DEFAULT NULL            COMMENT '最大阈值',
  `mid_th`      INT      DEFAULT NULL            COMMENT '中间阈值',
  `min_th`      INT      DEFAULT NULL            COMMENT '最小阈值',
  `create_time` DATETIME DEFAULT NULL            COMMENT '创建时间',
  `update_time` DATETIME DEFAULT NULL            COMMENT '更新时间',
  `delete_flag` TINYINT(1) DEFAULT 0            COMMENT '删除标志: 0-未删除; 1-已删除',
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='传感器配置表';

-- 传感器记录表
CREATE TABLE `edubox_sensor_record` (
  `id`       BIGINT      NOT NULL AUTO_INCREMENT COMMENT '主键，自增',
  `sensorId` BIGINT      DEFAULT NULL            COMMENT '传感器配置表中传感器Id',
  `com`      VARCHAR(64) DEFAULT NULL            COMMENT '通信模块',
  `value`    INT         DEFAULT NULL            COMMENT '当前值',
  `status`   INT         DEFAULT NULL            COMMENT '当前状态: -1-超出上下限; 0-正常; 1-轻微; 2-警告; 3-危险',
  `datetime` DATETIME    DEFAULT NULL            COMMENT '当前日期时间',
  PRIMARY KEY (`id`),
  KEY `idx_sensorId` (`sensorId`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='传感器记录表';
```
