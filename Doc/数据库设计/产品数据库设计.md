## 建表 SQL

```sql
-- 1. product_picture（图片表）
CREATE TABLE product_picture (
    id          BIGINT      NOT NULL AUTO_INCREMENT  COMMENT '主键，自增',
    content     LONGTEXT    DEFAULT NULL             COMMENT '图片内容（Base64/URL）',
    create_time DATETIME    DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='产品图片表';


-- 2. product_type（产品类型表）
CREATE TABLE product_type (
    id          BIGINT       NOT NULL AUTO_INCREMENT  COMMENT '主键，自增',
    name        VARCHAR(255) NOT NULL                 COMMENT '类型名称',
    code        VARCHAR(255) NOT NULL                 COMMENT '类型码',
    create_time DATETIME     DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    update_time DATETIME     DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
    delete_flag BOOLEAN      DEFAULT 0                COMMENT '删除标志，默认为0',
    PRIMARY KEY (id),
    UNIQUE INDEX uk_code (code)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='产品类型表';


-- 3. product_model（产品型号表）
CREATE TABLE product_model (
    id          BIGINT       NOT NULL AUTO_INCREMENT  COMMENT '主键，自增',
    name        VARCHAR(255) NOT NULL                 COMMENT '型号名称',
    code        VARCHAR(255) NOT NULL                 COMMENT '型号码',
    create_time DATETIME     DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    update_time DATETIME     DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
    delete_flag BOOLEAN      DEFAULT 0                COMMENT '删除标志，默认为0',
    PRIMARY KEY (id),
    UNIQUE INDEX uk_code (code)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='产品型号表';


-- 4. product_product（产品表）
CREATE TABLE product_product (
    id           BIGINT       NOT NULL AUTO_INCREMENT  COMMENT '主键，自增',
    bar_code     VARCHAR(255) DEFAULT NULL             COMMENT '条形码编号',
    type_code    VARCHAR(255) NOT NULL                 COMMENT '产品类型码',
    model_code   VARCHAR(255) NOT NULL                 COMMENT '产品型号码',
    product_code VARCHAR(255) DEFAULT NULL             COMMENT '产品代号',
    pic_id       BIGINT       DEFAULT NULL             COMMENT '图片id (已修改为BIGINT以匹配父表)',
    device_sn    VARCHAR(255) DEFAULT NULL             COMMENT '设备SN',
    create_time  DATETIME     DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    update_time  DATETIME     DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
    delete_flag  BOOLEAN      DEFAULT 0                COMMENT '删除标志，默认为0',
    PRIMARY KEY (id),
    UNIQUE INDEX uk_bar_code (bar_code),
    INDEX idx_type_code (type_code),
    INDEX idx_model_code (model_code),
    INDEX idx_device_sn (device_sn),
    CONSTRAINT fk_product_type FOREIGN KEY (type_code)  REFERENCES product_type(code),
    CONSTRAINT fk_product_model FOREIGN KEY (model_code) REFERENCES product_model(code),
    CONSTRAINT fk_product_pic   FOREIGN KEY (pic_id)     REFERENCES product_picture(id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='产品表';


-- 5. product_store（产品出入库表）
create table product_store
(
    id          bigint auto_increment comment '主键，自增'
        primary key,
    product_id  bigint            not null comment '产品id',
    action      tinyint default 0 not null comment '0:出库；1:入库',
    device_sn   varchar(255)      null comment '设备SN',
    group_id    bigint            null comment '分组id',
    employee_id bigint            null comment '入库员工id',
    datetime    datetime          null comment '入库时间',
    constraint fk_store_product
        foreign key (product_id) references product_product (id)
)
    comment '产品出入库表';

create index idx_action
    on product_store (action);

create index idx_device_sn
    on product_store (device_sn);

create index idx_group_id
    on product_store (group_id);

create index idx_in_employee_id
    on product_store (employee_id);

create index idx_product_id
    on product_store (product_id);




-- 6. product_production（产品产量表）
CREATE TABLE product_production (
    id                 BIGINT   NOT NULL AUTO_INCREMENT  COMMENT '主键，自增',
    product_id         BIGINT   NOT NULL                 COMMENT '产品id',
    planned_production INT      DEFAULT 0                COMMENT '计划产量',
    actual_production  INT      DEFAULT 0                COMMENT '实际产量',
    create_time        DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    store_num          INT      NOT NULL                 COMMENT '入库数量',
    qualified_num      INT      NOT NULL                 COMMENT '合格数量',
    group_id           BIGINT   NOT NULL                 COMMENT '分组id',
    plan_name          VARCHAR(63) NOT NULL              COMMENT '计划名称',
    PRIMARY KEY (id),
    INDEX idx_product_id (product_id),
    CONSTRAINT fk_production_product FOREIGN KEY (product_id) REFERENCES product_product(id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='产品产量表';
```
